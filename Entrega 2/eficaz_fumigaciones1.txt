drop database if exists eficaz_fumigaciones;
create database eficaz_fumigaciones;
use eficaz_fumigaciones;

create table clientes(
    idCliente int auto_increment primary key,
    tipoDocumento varchar(20) not null,
    numDocumento varchar(20) unique not null,
    nombreCliente varchar(100) not null,
    telefonoCliente varchar(20) null,
    celularCliente varchar(20) not null,
    correoCliente varchar(100) not null,
    estadoCliente varchar(20) default 'Activo'
);

create table propiedades(
    idPropiedad int auto_increment primary key,
    idClienteFK int not null,
    direccionPropiedad varchar(150) not null,
    tipoEspacio varchar(50) not null,
    areaM2 double not null
);

alter table propiedades
add constraint fk_clientes_propiedades
foreign key(idClienteFK)
references clientes(idCliente)
on update cascade
on delete cascade;

create table proveedor(
    idProveedor int auto_increment primary key,
    nit varchar(20) unique not null,
    razonSocial varchar(100) not null,
    telefonoProveedor varchar(20) null,
    correoProveedor varchar(100) null,
    ciudadProveedor varchar(50) null,
    estadoProveedor varchar(20) default 'Activo'
);

create table insumoquimico(
    idInsumoQuimico int auto_increment primary key,
    nombreInsumo varchar(100) not null,
    unidadMedida varchar(20) not null,
    stockActual double default 0.0,
    stockMinimo double not null,
    precioListaActual double not null,
    estadoInsumo varchar(20) default 'Activo'
);

create table plaga(
    idPlaga int auto_increment primary key,
    nombrePlaga varchar(100) not null,
    tarifaPorM2 double not null,
    dosisEstandarPorM2 double not null,
    idInsumoQuimicoFK int not null
);

alter table plaga
add constraint fk_plaga_insumo
foreign key(idInsumoQuimicoFK)
references insumoquimico(idInsumoQuimico);

create table citafumigacion(
    idCita int auto_increment primary key,
    fechaHora datetime not null,
    estadoCita varchar(20) default 'Pendiente',
    tipoSolicitud varchar(50) not null
);

create table visitacotizacion(
    idVisitaCotizacion int auto_increment primary key,
    valorServicio double default 0.0,
    tipoPropiedad varchar(50) null,
    areaM2 double null,
    direccionPropiedad varchar(150) null,
    idCitaFK int not null
);

alter table visitacotizacion
add constraint fk_visita_cita
foreign key(idCitaFK)
references citafumigacion(idCita)
on update cascade
on delete cascade;

create table serviciodirecto(
    idServicio int auto_increment primary key,
    costoTotal double not null,
    dosisInsumo double not null,
    estadoServicio varchar(20) default 'Realizado',
    fechaTransaccion datetime default current_timestamp,
    usuarioResponsable varchar(100) null,
    detalleOperacion varchar(255) null,
    idCitaFK int not null,
    idClienteFK int not null,
    idPlagaFK int not null
);

alter table serviciodirecto add constraint fk_servicio_cita foreign key(idCitaFK) references citafumigacion(idCita);
alter table serviciodirecto add constraint fk_servicio_cliente foreign key(idClienteFK) references clientes(idCliente);
alter table serviciodirecto add constraint fk_servicio_plaga foreign key(idPlagaFK) references plaga(idPlaga);

create table certificadofumigacion(
    idCertificado int auto_increment primary key,
    fechaEmision date not null,
    fechaVencimiento date not null,
    tokenEnlace varchar(30) unique not null,
    estadoVigencia varchar(20) default 'Vigente',
    idServicioFK int not null,
    idClienteFK int not null
);

alter table certificadofumigacion add constraint fk_certificado_servicio foreign key(idServicioFK) references serviciodirecto(idServicio);
alter table certificadofumigacion add constraint fk_certificado_cliente foreign key(idClienteFK) references clientes(idCliente);

create table ordencompra(
    idCompra int auto_increment primary key,
    fechaCompra date not null,
    valorTotal double not null,
    estadoCompra varchar(20) default 'Completada',
    fechaTransaccionCompra datetime default current_timestamp,
    usuarioResponsableCompra varchar(100) null,
    detalleCompra varchar(255) null,
    idProveedorFK int not null
);

alter table ordencompra add constraint fk_compra_proveedor foreign key(idProveedorFK) references proveedor(idProveedor);

create table detalleordencompra(
    idDetalle int auto_increment primary key,
    idCompraFK int not null,
    idInsumoQuimicoFK int not null,
    cantidad double not null,
    precioUnitario double not null,
    subtotal double not null
);

alter table detalleordencompra add constraint fk_detalle_compra foreign key(idCompraFK) references ordencompra(idCompra) on delete cascade;
alter table detalleordencompra add constraint fk_detalle_insumo foreign key(idInsumoQuimicoFK) references insumoquimico(idInsumoQuimico);

create table ajusteinventario(
    idAjuste int auto_increment primary key,
    fechaAjuste date not null,
    tipoAjuste varchar(50) not null,
    cantidad double not null,
    justificacion varchar(255) not null,
    fechaTransaccionAjuste datetime default current_timestamp,
    usuarioResponsable varchar(100) null,
    detalleAjuste varchar(255) null,
    idInsumoQuimicoFK int not null,
    idServicioFK int null
);

alter table ajusteinventario add constraint fk_ajuste_insumo foreign key(idInsumoQuimicoFK) references insumoquimico(idInsumoQuimico);
alter table ajusteinventario add constraint fk_ajuste_servicio foreign key(idServicioFK) references serviciodirecto(idServicio);

create view vista_clientes_propiedades as
select c.idCliente, c.nombreCliente, c.celularCliente, p.direccionPropiedad, p.tipoEspacio, p.areaM2
from clientes c
inner join propiedades p on c.idCliente = p.idClienteFK;

create view vista_inventario_critico as
select idInsumoQuimico, nombreInsumo, stockActual, stockMinimo, unidadMedida
from insumoquimico
where stockActual < stockMinimo;

create procedure sp_registrar_cliente(
    in p_tipoDoc varchar(20),
    in p_numDoc varchar(20),
    in p_nombre varchar(100),
    in p_tel varchar(20),
    in p_cel varchar(20),
    in p_correo varchar(100)
)
begin
    insert into clientes (tipoDocumento, numDocumento, nombreCliente, telefonoCliente, celularCliente, correoCliente)
    values (p_tipoDoc, p_numDoc, p_nombre, p_tel, p_cel, p_correo);
end //
Delimiter ;

Delimiter //
create procedure sp_consultar_stock(
    in p_idInsumo int,
    out p_stockActual double
)
begin
    select stockActual into p_stockActual
    from insumoquimico
    where idInsumoQuimico = p_idInsumo;
end //
Delimiter ;

Delimiter //
create trigger trg_actualizar_stock_compra
after insert on detalleordencompra
for each row
begin
    update insumoquimico
    set stockActual = stockActual + new.cantidad
    where idInsumoQuimico = new.idInsumoQuimicoFK;
end //
Delimiter ;

Delimiter //
create trigger trg_validar_stock_servicio
before insert on serviciodirecto
for each row
begin
    declare v_stock double;
    select i.stockActual into v_stock
    from insumoquimico i
    inner join plaga p on i.idInsumoQuimico = p.idInsumoQuimicoFK
    where p.idPlaga = new.idPlagaFK;
    if v_stock < new.dosisInsumo then
        signal sqlstate '45000'
        set message_text = 'Stock insuficiente del insumo';
    end if;
end //
Delimiter ;

Delimiter //
create trigger trg_descontar_insumo_servicio
after insert on serviciodirecto
for each row
begin
    update insumoquimico i
    inner join plaga p on i.idInsumoQuimico = p.idInsumoQuimicoFK
    set i.stockActual = i.stockActual - new.dosisInsumo
    where p.idPlaga = new.idPlagaFK;
end //
Delimiter ;
