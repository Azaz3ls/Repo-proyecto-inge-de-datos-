// VISTAS
db.createView("Vista_ResumenServicios", "citas_fumigacion", [
  { $match: { "servicioDirecto": { $exists: true } } },
  { $lookup: { from: "clientes", localField: "idCliente", foreignField: "_id", as: "cli" } },
  { $unwind: "$cli" },
  { $lookup: { from: "plagas", localField: "servicioDirecto.idPlaga", foreignField: "_id", as: "plaga" } },
  { $unwind: "$plaga" },
  { $project: { _id: 0, idServicio: "$servicioDirecto.idServicio", nombreCliente: "$cli.nombreCliente", celularCliente: "$cli.celularCliente", nombrePlaga: "$plaga.nombrePlaga", costoTotal: "$servicioDirecto.costoTotal", tokenEnlace: "$servicioDirecto.certificado.tokenEnlace", estadoVigencia: "$servicioDirecto.certificado.estadoVigencia" } }
]);

db.createView("Vista_AlertasInventario", "insumos_quimicos", [
  { $match: { $expr: { $lte: ["$stockActual", "$stockMinimo"] } } },
  { $addFields: { alerta: { $cond: [{ $eq: ["$stockActual", 0] }, "AGOTADO", "STOCK BAJO"] } } }
]);

db.createView("Vista_AgendaVisitas", "citas_fumigacion", [
  { $match: { tipoSolicitud: "Visita Cotización" } },
  { $lookup: { from: "clientes", localField: "idCliente", foreignField: "_id", as: "cli" } },
  { $unwind: "$cli" },
  { $lookup: { from: "usuarios", localField: "idUsuario", foreignField: "_id", as: "tec" } },
  { $unwind: "$tec" },
  { $project: { _id: 0, idCita: "$_id", fechaHora: 1, estadoCita: 1, nombreCliente: "$cli.nombreCliente", tecnicoAsignado: "$tec.nombreUsuario", areaMedidaM2: "$visitaCotizacion.areaMedidaM2", valorServicio: "$visitaCotizacion.valorServicio", estadoCotizacion: "$visitaCotizacion.estadoCotizacion" } }
]);

  { $lookup: { from: "usuarios", localField: "idUsuario", foreignField: "_id", as: "usr" } },
  { $unwind: "$usr" },
  { $project: { _id: 0, modulo: 1, idRegistro: 1, accion: 1, valorAnterior: 1, valorNuevo: 1, fechaEvento: 1, nombreUsuario: "$usr.nombreUsuario" } }
]);
