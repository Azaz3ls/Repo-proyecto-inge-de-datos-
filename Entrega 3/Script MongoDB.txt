db.dropDatabase();

db.createCollection('usuarios', {
  validator: {
    $jsonSchema: {
      bsonType: 'object',
      required: ['_id', 'nombreUsuario', 'correoUsuario', 'rolUsuario', 'estadoUsuario'],
      properties: {
        _id: { bsonType: 'int' },
        nombreUsuario: { bsonType: 'string' },
        correoUsuario: { bsonType: 'string', pattern: '^.+@.+\\..+$' },
        rolUsuario: { enum: ['Sistema', 'Administrador', 'Técnico', 'Inventario', 'Compras'] },
        estadoUsuario: { enum: ['Activo', 'Inactivo'] }
      }
    }
  }
});
db.usuarios.createIndex({ correoUsuario: 1 }, { unique: true });

db.createCollection('clientes', {
  validator: {
    $jsonSchema: {
      bsonType: 'object',
      required: ['_id', 'tipoDocumento', 'numDocumento', 'nombreCliente', 'celularCliente', 'correoCliente', 'estadoCliente', 'propiedades'],
      properties: {
        _id: { bsonType: 'int' },
        tipoDocumento: { enum: ['CC', 'CE', 'NIT', 'PAS'] },
        numDocumento: { bsonType: 'string' },
        nombreCliente: { bsonType: 'string' },
        celularCliente: { bsonType: 'string', pattern: '^[0-9]{7,15}$' },
        telefonoCliente: { bsonType: ['string', 'null'] },
        correoCliente: { bsonType: 'string', pattern: '^.+@.+\\..+$' },
        estadoCliente: { enum: ['Activo', 'Inactivo'] },
        propiedades: {
          bsonType: 'array',
          items: {
            bsonType: 'object',
            required: ['idPropiedad', 'tipoPropiedad', 'direccionPropiedad', 'estadoPropiedad'],
            properties: {
              idPropiedad: { bsonType: 'int' },
              tipoPropiedad: { enum: ['Casa', 'Apartamento', 'Local Comercial', 'Oficina', 'Bodega', 'Hotel', 'Restaurante', 'Institución Educativa'] },
              direccionPropiedad: { bsonType: 'string' },
              areaM2: { bsonType: ['double', 'int', 'null'], minimum: 0.01 },
              estadoPropiedad: { enum: ['Activa', 'Inactiva'] }
            }
          }
        }
      }
    }
  }
});
db.clientes.createIndex({ numDocumento: 1 }, { unique: true });
db.clientes.createIndex({ correoCliente: 1 }, { unique: true });
db.clientes.createIndex({ nombreCliente: 1 });
db.clientes.createIndex({ 'propiedades.idPropiedad': 1 }, { unique: true });

db.createCollection('insumos_quimicos', {

        _id: { bsonType: 'int' },
        nombreInsumo: { bsonType: 'string' },
        unidadMedida: { bsonType: 'string' },
        stockActual: { bsonType: ['double', 'int'], minimum: 0 },

db.createView('Vista_AutonomiaInventario', 'insumos_quimicos', [
  {
    $lookup: {
      from: 'ajustes_inventario',
      let: { idInsumo: '$_id' },
      pipeline: [
        {
          $match: {
            $expr: {
              $and: [
                { $eq: ['$idInsumoQuimico', '$$idInsumo'] },
                { $eq: ['$tipoAjuste', 'Consumo Servicio'] },
                { $gte: ['$fechaAjuste', new Date(Date.now() - 90 * 24 * 60 * 60 * 1000)] }
              ]
            }
          }
        }
      ],
      as: 'consumos'
    }
  },
  {
    $project: {
      nombreInsumo: 1,
      unidadMedida: 1,
      stockActual: 1,
      consumo90Dias: { $sum: '$consumos.cantidad' },
      consumoMensualPromedio: { $round: [{ $divide: [{ $sum: '$consumos.cantidad' }, 3] }, 2] },
      diasAutonomia: {
        $cond: [
          { $eq: [{ $sum: '$consumos.cantidad' }, 0] },
          null,
          { $floor: { $divide: ['$stockActual', { $divide: [{ $sum: '$consumos.cantidad' }, 90] }] } }
        ]
      }
    }
  }
]);
