--  Ajustes de esta versión (cierre de requisitos funcionales pendientes):
--   - Tabla Propiedad: un cliente puede tener varios inmuebles (RQF 004-006).
--     La cita se agenda sobre una propiedad, y por ella se llega al cliente
--     (Cita -> Propiedad -> Cliente), así el modelo sigue sin ciclos.
--   - Restricciones CHECK en estados, tipos de solicitud, tipos de propiedad,
--     correo y teléfonos. Correo del cliente único (acceso sin contraseña).
--   - Proveedor con razón social. Catálogo por proveedor como vista.
--   - Validación de cruces por rango horario, reprogramación de citas,
--     control de cambios de estado y certificado automático.
--   - Reportes: certificados por vencer, ganancia neta, ranking y autonomía.
--
--  Modelo sin ciclos (cada línea = una FK; el grafo queda como un árbol):
--     Clientes ── Propiedad ── CitaFumigacion ── Usuarios
--     CitaFumigacion ── VisitaCotizacion
--     CitaFumigacion ── ServicioDirecto ── Plaga ── InsumoQuimico
--     ServicioDirecto ── CertificadoFumigacion
--     InsumoQuimico ── AjusteInventario
--     InsumoQuimico ── DetalleOrdenCompra ── OrdenCompra ── Proveedor
--     Usuarios ── Auditoria* (4 tablas hoja; el idRegistro NO es FK, así
--                            el historial sobrevive aunque se borre el dato)
-- =====================================================================

DROP DATABASE IF EXISTS eficaz_fumigaciones;
CREATE DATABASE eficaz_fumigaciones;
USE eficaz_fumigaciones;

-- ---------------------------------------------------------------------
-- Usuarios
-- ---------------------------------------------------------------------
create table Usuarios (
    idUsuario int auto_increment primary key,
    nombreUsuario varchar(100) not null,
    correoUsuario varchar(100) not null unique,
    rolUsuario varchar(30) not null default 'Técnico'
        check (rolUsuario in ('Sistema','Administrador','Técnico','Inventario','Compras')),
    estadoUsuario varchar(20) not null default 'Activo'
        check (estadoUsuario in ('Activo','Inactivo')),
    fechaCreacion datetime not null default current_timestamp
);

-- Clientes: se agrega teléfono fijo (opcional), el correo pasa a ser único
-- (sirve para el acceso sin contraseña) y se validan formatos.
create table Clientes (
    idCliente int auto_increment primary key,
    tipoDocumento varchar(20) not null
        check (tipoDocumento in ('CC','CE','NIT','PAS')),
    numDocumento varchar(20) not null unique,
    nombreCliente varchar(100) not null,
    celularCliente varchar(20) not null
        check (celularCliente regexp '^[0-9]{7,15}$'),
    telefonoCliente varchar(20)
        check (telefonoCliente is null or telefonoCliente regexp '^[0-9]{7,15}$'),
    correoCliente varchar(100) not null unique
        check (correoCliente like '%_@_%._%'),
    estadoCliente varchar(20) not null default 'Activo'
        check (estadoCliente in ('Activo','Inactivo'))
);
create index idx_clientes_nombre on Clientes(nombreCliente);
create index idx_clientes_celular on Clientes(celularCliente);

-- TABLA NUEVA: Propiedad (varios inmuebles por cliente)
-- areaM2 puede quedar vacía cuando el cliente no conoce los metros
-- (en ese caso se agenda una visita de cotización para medir).
create table Propiedad (
    idPropiedad int auto_increment primary key,
    tipoPropiedad varchar(50) not null
        check (tipoPropiedad in ('Casa','Apartamento','Local Comercial','Oficina','Bodega',
                                 'Hotel','Restaurante','Institución Educativa')),
    direccionPropiedad varchar(150) not null,
    areaM2 decimal(10,2) check (areaM2 is null or areaM2 > 0),
    estadoPropiedad varchar(20) not null default 'Activa'
        check (estadoPropiedad in ('Activa','Inactiva')),
    idCliente int not null,
    foreign key (idCliente) references Clientes(idCliente)
);

-- CitaFumigacion: se agenda sobre una propiedad (de ahí sale el cliente) y
-- con un técnico (Usuarios).
create table CitaFumigacion (
    idCita int auto_increment primary key,
    fechaHora datetime not null,
    estadoCita varchar(20) not null default 'Pendiente'
        check (estadoCita in ('Pendiente','En Camino','Realizado','Cancelado')),
    tipoSolicitud varchar(50) not null
        check (tipoSolicitud in ('Visita Cotización','Servicio Directo')),
    duracionEstimadaMin int not null default 60 check (duracionEstimadaMin > 0),
    observaciones varchar(255),
    fechaCreacion datetime not null default current_timestamp,
    idPropiedad int not null,
    idUsuario int not null,
    foreign key (idPropiedad) references Propiedad(idPropiedad),
    foreign key (idUsuario) references Usuarios(idUsuario)
);

-- VisitaCotizacion: nace en $0 y 'Por inspeccionar'; el área medida y el
-- valor se llenan después de la inspección (sp_CotizarVisita).
create table VisitaCotizacion (
    idVisitaCotizacion int auto_increment primary key,
    valorServicio decimal(10,2) not null default 0.00 check (valorServicio >= 0),
    areaMedidaM2 decimal(10,2) check (areaMedidaM2 is null or areaMedidaM2 > 0),
    estadoCotizacion varchar(20) not null default 'Por inspeccionar'
        check (estadoCotizacion in ('Por inspeccionar','Cotizada')),
    observacionInspeccion varchar(255),
    idCita int not null unique,
    foreign key (idCita) references CitaFumigacion(idCita) on delete cascade
);

create table InsumoQuimico (
    idInsumoQuimico int auto_increment primary key,
    nombreInsumo varchar(100) not null unique,
    unidadMedida varchar(20) not null,
    stockActual decimal(10,2) not null check (stockActual >= 0),
    stockMinimo decimal(10,2) not null check (stockMinimo >= 0),
    precioListaActual decimal(10,2) not null check (precioListaActual > 0),
    estadoInsumo varchar(20) not null default 'Disponible'
);

create table Plaga (
    idPlaga int auto_increment primary key,
    nombrePlaga varchar(100) not null unique,
    tarifaPorM2 decimal(10,2) not null check (tarifaPorM2 > 0),
    dosisEstandarPorM2 decimal(10,2) not null check (dosisEstandarPorM2 > 0),
    idInsumoQuimico int not null,
    foreign key (idInsumoQuimico) references InsumoQuimico(idInsumoQuimico)
);

-- ServicioDirecto: guarda también el área con la que se calculó el cobro
-- (costo y dosis quedan congelados aunque cambie la tarifa de la plaga).
create table ServicioDirecto (
    idServicio int auto_increment primary key,
    costoTotal decimal(10,2) not null check (costoTotal >= 0),
    dosisInsumo decimal(10,2) not null check (dosisInsumo >= 0),
    areaM2 decimal(10,2) not null check (areaM2 > 0),
    estadoServicio boolean not null default true,
    fechaTransaccion datetime not null default current_timestamp,
    detalleOperacion varchar(255),
    idCita int not null unique,                  -- una cita genera un solo servicio
    idPlaga int not null,
    foreign key (idCita) references CitaFumigacion(idCita),
    foreign key (idPlaga) references Plaga(idPlaga)
);

create table CertificadoFumigacion (
    idCertificado int auto_increment primary key,
    fechaEmision date not null,
    fechaVencimiento date not null,
    tokenEnlace varchar(30) not null unique check (char_length(tokenEnlace) = 30),
    estadoVigencia varchar(20) not null default 'Vigente'
        check (estadoVigencia in ('Vigente','Vencido')),
    idServicio int not null,
    foreign key (idServicio) references ServicioDirecto(idServicio)
);

create table AjusteInventario (
    idAjuste int auto_increment primary key,
    fechaAjuste date not null,
    tipoAjuste varchar(50) not null
        check (tipoAjuste in ('Entrada','Compra Recibida','Salida','Consumo Servicio','Merma')),
    cantidad decimal(10,2) not null check (cantidad > 0),
    justificacion varchar(255) not null check (char_length(trim(justificacion)) > 0),
    fechaTransaccionAjuste datetime not null default current_timestamp,
    detalleAjuste varchar(255),
    idInsumoQuimico int not null,
    foreign key (idInsumoQuimico) references InsumoQuimico(idInsumoQuimico)
);

-- Proveedor: se agrega la razón social
create table Proveedor (
    idProveedor int auto_increment primary key,
    nit varchar(20) not null unique,
    razonSocial varchar(100) not null,
    telefonoProveedor varchar(20) not null,
    correoProveedor varchar(100) not null,
    ciudadProveedor varchar(50) not null,
    estadoProveedor varchar(20) not null default 'Activo'
        check (estadoProveedor in ('Activo','Inactivo'))
);

create table OrdenCompra (
    idCompra int auto_increment primary key,
    fechaCompra date not null,
    valorTotal decimal(10,2) not null default 0.00 check (valorTotal >= 0),
    estadoCompra varchar(20) not null default 'Pendiente'
        check (estadoCompra in ('Pendiente','Recibida','Cancelada')),
    fechaTransaccionCompra datetime not null default current_timestamp,
    detalleCompra varchar(255),
    idProveedor int not null,
    foreign key (idProveedor) references Proveedor(idProveedor)
);

create table DetalleOrdenCompra (
    idDetalle int auto_increment primary key,
    cantidad decimal(10,2) not null check (cantidad > 0),
    precioUnitario decimal(10,2) not null check (precioUnitario > 0),
    idCompra int not null,
    idInsumoQuimico int not null,
    unique (idCompra, idInsumoQuimico),
    foreign key (idCompra) references OrdenCompra(idCompra) on delete cascade,
    foreign key (idInsumoQuimico) references InsumoQuimico(idInsumoQuimico)
);

-- ---------------------------------------------------------------------
-- TABLAS DE AUDITORÍA (hojas del modelo: solo apuntan a Usuarios)
-- idRegistro guarda el id del dato auditado SIN foreign key a propósito.
-- ---------------------------------------------------------------------
create table AuditoriaClientes (
    idAuditoria int auto_increment primary key,
    idRegistro int not null,
    accion varchar(20) not null,
    valorAnterior varchar(500),
    valorNuevo varchar(500),
    fechaEvento datetime not null default current_timestamp,
    idUsuario int not null,
    foreign key (idUsuario) references Usuarios(idUsuario)
);

create table AuditoriaInventario (
    idAuditoria int auto_increment primary key,
    idRegistro int not null,
    accion varchar(20) not null,
    valorAnterior varchar(500),
    valorNuevo varchar(500),
    fechaEvento datetime not null default current_timestamp,
    idUsuario int not null,
    foreign key (idUsuario) references Usuarios(idUsuario)
);

create table AuditoriaServicios (
    idAuditoria int auto_increment primary key,
    idRegistro int not null,
    accion varchar(20) not null,
    valorAnterior varchar(500),
    valorNuevo varchar(500),
    fechaEvento datetime not null default current_timestamp,
    idUsuario int not null,
    foreign key (idUsuario) references Usuarios(idUsuario)
);

create table AuditoriaCompras (
    idAuditoria int auto_increment primary key,
    idRegistro int not null,
    accion varchar(20) not null,
    valorAnterior varchar(500),
    valorNuevo varchar(500),
    fechaEvento datetime not null default current_timestamp,
    idUsuario int not null,
    foreign key (idUsuario) references Usuarios(idUsuario)
);

USE eficaz_fumigaciones;

-- 0. TABLA: Usuarios

select * from Usuarios;

select rolUsuario, count(*) as TotalUsuarios
from Usuarios
group by rolUsuario;

-- 1. TABLA: Clientes

select * from Clientes;

select idCliente, numDocumento, nombreCliente, celularCliente, correoCliente
from Clientes
where tipoDocumento = 'CC' and estadoCliente = 'Activo';

select tipoDocumento, count(*) as TotalClientes
from Clientes
group by tipoDocumento;

select * from Clientes
where numDocumento = '1018400111';


-- 2. TABLA: Propiedad (nueva)

select * from Propiedad;

-- Clientes con más de una propiedad registrada
select cli.idCliente, cli.nombreCliente, count(p.idPropiedad) as TotalPropiedades
from Clientes cli
inner join Propiedad p on p.idCliente = cli.idCliente
group by cli.idCliente, cli.nombreCliente
having count(p.idPropiedad) > 1;

select tipoPropiedad, count(*) as Cantidad, avg(areaM2) as PromedioAreaM2
from Propiedad
group by tipoPropiedad;


-- 3. TABLA: CitaFumigacion

select * from CitaFumigacion;

select idCita, fechaHora, tipoSolicitud
from CitaFumigacion
where estadoCita = 'Realizado';

select tipoSolicitud, count(*) as CantidadCitas
from CitaFumigacion
group by tipoSolicitud;

select * from CitaFumigacion
order by fechaHora desc;

from CitaFumigacion c

-- Autonomía del inventario: consumo promedio mensual (últimos 90 días)
-- y días que alcanza el stock actual a ese ritmo.
create or replace view Vista_AutonomiaInventario as
select
    i.idInsumoQuimico,
    i.nombreInsumo,
    i.unidadMedida,
    i.stockActual,
    round(coalesce(sum(a.cantidad), 0) / 3, 2) as consumoMensualPromedio,
    case
        when coalesce(sum(a.cantidad), 0) = 0 then null
        else floor(i.stockActual / (sum(a.cantidad) / 90))
    end as diasAutonomia
from InsumoQuimico i
left join AjusteInventario a
       on a.idInsumoQuimico = i.idInsumoQuimico
      and a.tipoAjuste = 'Consumo Servicio'
      and a.fechaAjuste >= date_sub(curdate(), interval 90 day)
group by i.idInsumoQuimico, i.nombreInsumo, i.unidadMedida, i.stockActual;


-- =====================================================================
--  FUNCIÓN DE DISPONIBILIDAD
--  Devuelve true si el técnico está libre en el rango [inicio, inicio + duración).
--  Se usa en los triggers de citas (agendar y reprogramar).
-- =====================================================================
USE eficaz_fumigaciones;

delimiter //
create function fn_TecnicoDisponible(
    p_idUsuario int,
    p_inicio datetime,
    p_duracionMin int,
    p_idCitaExcluir int
) returns boolean
reads sql data
begin
    return not exists (
        select 1
        from CitaFumigacion c
        where c.idUsuario = p_idUsuario
          and c.estadoCita in ('Pendiente', 'En Camino')
          and c.idCita <> coalesce(p_idCitaExcluir, 0)
          and p_inicio < date_add(c.fechaHora, interval c.duracionEstimadaMin minute)
          and date_add(p_inicio, interval p_duracionMin minute) > c.fechaHora
    );
end //
delimiter ;


-- =====================================================================
--  CREACIÓN DE PROCEDIMIENTOS ALMACENADOS
--  Todos reciben p_idUsuario: se guarda en la variable de sesión
--  @idUsuarioActual, que los triggers leen para llenar las tablas de
--  auditoría. Si nadie la define, los triggers usan el usuario 1 (Sistema).
-- =====================================================================
USE eficaz_fumigaciones;

-- Procedimiento auxiliar: valida el usuario y lo deja como "usuario actual"
delimiter //
create procedure sp_ValidarUsuario(in p_idUsuario int)
begin
    if not exists (select 1 from Usuarios where idUsuario = p_idUsuario and estadoUsuario = 'Activo') then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: El usuario no existe o está inactivo.';
    end if;
    set @idUsuarioActual = p_idUsuario;
end //
delimiter ;


-- SP 1: Registrar usuario
delimiter //
create procedure sp_RegistrarUsuario(
    in p_nombre varchar(100),
    in p_correo varchar(100),
    in p_rol varchar(30)
)
begin
    if exists (select 1 from Usuarios where correoUsuario = p_correo) then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: Ya existe un usuario con ese correo.';
    else
        insert into Usuarios (nombreUsuario, correoUsuario, rolUsuario)
        values (p_nombre, p_correo, p_rol);
    end if;
end //
delimiter ;


-- SP 2: Registrar cliente (documento y correo únicos)
delimiter //
create procedure sp_RegistrarCliente(
    in p_tipoDoc varchar(20),
    in p_numDoc varchar(20),
    in p_nombre varchar(100),
    in p_celular varchar(20),
    in p_telefono varchar(20),
    in p_correo varchar(100),
    in p_idUsuario int
)
begin
    call sp_ValidarUsuario(p_idUsuario);

    if exists (select 1 from Clientes where numDocumento = p_numDoc) then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: El número de documento ya se encuentra registrado.';
    end if;
    if exists (select 1 from Clientes where correoCliente = p_correo) then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: El correo ya está registrado por otro cliente.';
    end if;

    insert into Clientes (tipoDocumento, numDocumento, nombreCliente, celularCliente, telefonoCliente, correoCliente, estadoCliente)
    values (p_tipoDoc, p_numDoc, p_nombre, p_celular, p_telefono, p_correo, 'Activo');

    set @idUsuarioActual = null;
end //
delimiter ;


-- SP 3: Actualizar datos de contacto (el documento no se puede cambiar)
delimiter //
create procedure sp_ActualizarContactoCliente(
    in p_idCliente int,
    in p_celular varchar(20),
    in p_telefono varchar(20),
    in p_correo varchar(100),
    in p_idUsuario int
)
begin
    call sp_ValidarUsuario(p_idUsuario);

    if not exists (select 1 from Clientes where idCliente = p_idCliente) then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: El cliente no existe.';
    end if;
    if exists (select 1 from Clientes where correoCliente = p_correo and idCliente <> p_idCliente) then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: El correo ya está registrado por otro cliente.';
    end if;

    update Clientes
    set celularCliente = coalesce(p_celular, celularCliente),
        telefonoCliente = coalesce(p_telefono, telefonoCliente),
        correoCliente = coalesce(p_correo, correoCliente)
    where idCliente = p_idCliente;

    set @idUsuarioActual = null;
end //
delimiter ;


-- SP 4: Activar / inactivar cliente (los clientes NO se borran, se inactivan)
delimiter //
create procedure sp_CambiarEstadoCliente(
    in p_idCliente int,
    in p_nuevoEstado varchar(20),
    in p_idUsuario int
)
begin
    call sp_ValidarUsuario(p_idUsuario);

    if p_nuevoEstado not in ('Activo', 'Inactivo') then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: El estado debe ser Activo o Inactivo.';
    end if;
    if not exists (select 1 from Clientes where idCliente = p_idCliente) then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: El cliente no existe.';
    end if;

    update Clientes set estadoCliente = p_nuevoEstado where idCliente = p_idCliente;
    set @idUsuarioActual = null;
end //
delimiter ;


-- SP 5: Acceso del cliente sin contraseña (solo con el correo registrado)
delimiter //
create procedure sp_AccesoCliente(in p_correo varchar(100))
begin
    if not exists (select 1 from Clientes where correoCliente = trim(p_correo) and estadoCliente = 'Activo') then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Acceso denegado: el correo no corresponde a un cliente activo.';
    end if;

    select idCliente, nombreCliente, tipoDocumento, numDocumento, celularCliente, correoCliente
    from Clientes
    where correoCliente = trim(p_correo);
end //
delimiter ;


-- SP 6: Buscador de clientes por nombre, documento o teléfono
delimiter //
create procedure sp_BuscarCliente(in p_texto varchar(100))
begin
    select idCliente, tipoDocumento, numDocumento, nombreCliente, celularCliente, telefonoCliente, correoCliente, estadoCliente
    from Clientes
    where nombreCliente like concat('%', p_texto, '%')
       or numDocumento like concat(p_texto, '%')
       or celularCliente like concat(p_texto, '%')
       or telefonoCliente like concat(p_texto, '%')
    order by nombreCliente;
end //
delimiter ;


-- SP 7: Registrar propiedad de un cliente
delimiter //
create procedure sp_RegistrarPropiedad(
    in p_idCliente int,
    in p_tipoPropiedad varchar(50),
    in p_direccion varchar(150),
    in p_areaM2 decimal(10,2),
    in p_idUsuario int,
    out p_idPropiedad int
)
begin
    call sp_ValidarUsuario(p_idUsuario);

    if not exists (select 1 from Clientes where idCliente = p_idCliente and estadoCliente = 'Activo') then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: El cliente no existe o está inactivo.';
    end if;
    if p_areaM2 is not null and p_areaM2 <= 0 then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: El área debe ser mayor a cero.';
    end if;

    insert into Propiedad (tipoPropiedad, direccionPropiedad, areaM2, idCliente)
    values (p_tipoPropiedad, p_direccion, p_areaM2, p_idCliente);

    set p_idPropiedad = last_insert_id();
    set @idUsuarioActual = null;
end //
delimiter ;

-- SP 12: Cotizar una visita después de la inspección
-- (la visita se crea sola en $0 al agendar; aquí se registra lo medido)
delimiter //
create procedure sp_CotizarVisita(
    in p_idCita int,
    in p_areaMedidaM2 decimal(10,2),
    in p_idPlaga int,
    in p_observacion varchar(255),
    in p_idUsuario int
)
begin
    declare v_tarifa decimal(10,2);

    call sp_ValidarUsuario(p_idUsuario);

    if not exists (select 1 from VisitaCotizacion where idCita = p_idCita and estadoCotizacion = 'Por inspeccionar') then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: La cita no tiene una visita pendiente por cotizar.';
    end if;
    if p_areaMedidaM2 is null or p_areaMedidaM2 <= 0 then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: El área medida debe ser mayor a cero.';
    end if;

    select tarifaPorM2 into v_tarifa from Plaga where idPlaga = p_idPlaga;
    if v_tarifa is null then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: La plaga indicada no existe.';
    end if;

    update VisitaCotizacion
    set areaMedidaM2 = p_areaMedidaM2,
        valorServicio = p_areaMedidaM2 * v_tarifa,
        estadoCotizacion = 'Cotizada',
        observacionInspeccion = p_observacion
    where idCita = p_idCita;

    -- El área medida queda guardada en la propiedad para futuros servicios
    update Propiedad p
    inner join CitaFumigacion c on c.idPropiedad = p.idPropiedad
    set p.areaM2 = p_areaMedidaM2
    where c.idCita = p_idCita;

    update CitaFumigacion set estadoCita = 'Realizado' where idCita = p_idCita;
    set @idUsuarioActual = null;
end //
delimiter ;


-- SP 13: Registrar servicio directo (calcula costo y dosis; los triggers hacen el resto)
-- Si no se envía el área, se toma la registrada en la propiedad.
delimiter //
create procedure sp_RegistrarServicio(
    in p_idCita int,
    in p_idPlaga int,
    in p_areaM2 decimal(10,2),
    in p_idUsuario int,
    in p_detalle varchar(255),
    out p_idServicio int
)
begin
    declare v_tarifa decimal(10,2);
    declare v_dosis decimal(10,2);
    declare v_area decimal(10,2);

    call sp_ValidarUsuario(p_idUsuario);

    set v_area = p_areaM2;
    if v_area is null then
        select p.areaM2 into v_area
        from CitaFumigacion c
        inner join Propiedad p on c.idPropiedad = p.idPropiedad
        where c.idCita = p_idCita;
    end if;

    if v_area is null or v_area <= 0 then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: Se debe indicar el área en m2 (mayor a cero).';
    end if;
    if exists (select 1 from ServicioDirecto where idCita = p_idCita) then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: Esa cita ya tiene un servicio registrado.';
    end if;

    select tarifaPorM2, dosisEstandarPorM2 into v_tarifa, v_dosis from Plaga where idPlaga = p_idPlaga;
    if v_tarifa is null then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: Se debe indicar una plaga válida.';
    end if;

    insert into ServicioDirecto (costoTotal, dosisInsumo, areaM2, estadoServicio, detalleOperacion, idCita, idPlaga)
    values (v_area * v_tarifa, v_area * v_dosis, v_area, true, p_detalle, p_idCita, p_idPlaga);

    set p_idServicio = last_insert_id();
    set @idUsuarioActual = null;
end //
delimiter ;


-- SP 14: Emitir (o renovar) certificado de fumigación de forma manual
delimiter //
create procedure sp_EmitirCertificado(
    in p_idServicio int,
    in p_mesesVigencia int,
    in p_idUsuario int,
    out p_token varchar(30)
)
begin
    call sp_ValidarUsuario(p_idUsuario);

    if exists (select 1 from CertificadoFumigacion
               where idServicio = p_idServicio and estadoVigencia = 'Vigente') then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: El servicio ya tiene un certificado vigente.';
    end if;

    set p_token = upper(left(sha2(concat(uuid(), rand()), 256), 30));

    insert into CertificadoFumigacion (fechaEmision, fechaVencimiento, tokenEnlace, estadoVigencia, idServicio)
    values (curdate(), date_add(curdate(), interval p_mesesVigencia month), p_token, 'Vigente', p_idServicio);

    set @idUsuarioActual = null;
end //

-- SP 17: Actualizar stock (positivo = entra, negativo = sale); siempre deja un ajuste
delimiter //
create procedure sp_ActualizarStockInsumo(
    in p_idInsumo int,
    in p_cantidadAjuste decimal(10,2),
    in p_justificacion varchar(255),
    in p_idUsuario int
)
begin
    if p_cantidadAjuste = 0 then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: La cantidad del ajuste no puede ser cero.';
    elseif p_cantidadAjuste > 0 then
        call sp_RegistrarAjusteInventario(p_idInsumo, 'Entrada', p_cantidadAjuste, p_justificacion, p_idUsuario);
    else
        call sp_RegistrarAjusteInventario(p_idInsumo, 'Salida', abs(p_cantidadAjuste), p_justificacion, p_idUsuario);
    end if;
end //

-- SP 21: Agregar insumo a una orden. Si no se da precio, usa el del catálogo
-- del proveedor y, si no hay, el precio de lista del insumo.
delimiter //
create procedure sp_AgregarDetalleOrden(
    in p_idCompra int,
    in p_idInsumo int,
    in p_cantidad decimal(10,2),
    in p_precioUnitario decimal(10,2),
    in p_idUsuario int
)
begin
    declare v_precio decimal(10,2);

    call sp_ValidarUsuario(p_idUsuario);

    if exists (select 1 from DetalleOrdenCompra where idCompra = p_idCompra and idInsumoQuimico = p_idInsumo) then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: Ese insumo ya está en la orden.';
    end if;

    set v_precio = p_precioUnitario;
    if v_precio is null then
        select vc.precioVigente into v_precio
        from Vista_CatalogoProveedor vc
        inner join OrdenCompra oc on oc.idProveedor = vc.idProveedor
        where oc.idCompra = p_idCompra and vc.idInsumoQuimico = p_idInsumo;
    end if;
    if v_precio is null then
        select precioListaActual into v_precio from InsumoQuimico where idInsumoQuimico = p_idInsumo;
    end if;

    insert into DetalleOrdenCompra (cantidad, precioUnitario, idCompra, idInsumoQuimico)
    values (p_cantidad, v_precio, p_idCompra, p_idInsumo);

    set @idUsuarioActual = null;
end //
delimiter ;

-- SP 24: Reporte de servicios en un rango de fechas
delimiter //
create procedure sp_ReporteServiciosPorFecha(
    in p_fechaInicio date,
    in p_fechaFin date
)
begin
    select
        s.idServicio,
        s.fechaTransaccion,
        cli.nombreCliente,
        pr.direccionPropiedad,
        p.nombrePlaga,
        u.nombreUsuario as tecnico,
        s.costoTotal
    from ServicioDirecto s
    inner join CitaFumigacion cit on s.idCita = cit.idCita
    inner join Propiedad pr on cit.idPropiedad = pr.idPropiedad
    inner join Clientes cli on pr.idCliente = cli.idCliente
    inner join Usuarios u on cit.idUsuario = u.idUsuario
    inner join Plaga p on s.idPlaga = p.idPlaga
    where date(s.fechaTransaccion) between p_fechaInicio and p_fechaFin
    order by s.fechaTransaccion;
end //

-- SP 25: Historial completo de un cliente (de la más reciente a la más antigua)
-- con plaga tratada, químico aplicado, costo y certificado.
delimiter //
create procedure sp_HistorialCliente(in p_idCliente int)
begin
    select
        cit.idCita,
        cit.fechaHora,
        cit.estadoCita,
        cit.tipoSolicitud,
        pr.direccionPropiedad,
        s.idServicio,
        p.nombrePlaga as plagaTratada,
        i.nombreInsumo as quimicoAplicado,
        s.dosisInsumo,
        i.unidadMedida,
        s.costoTotal,
        cert.tokenEnlace,
        cert.estadoVigencia
    from CitaFumigacion cit
    inner join Propiedad pr on cit.idPropiedad = pr.idPropiedad
    left join ServicioDirecto s on s.idCita = cit.idCita
    left join Plaga p on s.idPlaga = p.idPlaga
    left join InsumoQuimico i on p.idInsumoQuimico = i.idInsumoQuimico
    left join CertificadoFumigacion cert on cert.idServicio = s.idServicio
    where pr.idCliente = p_idCliente
    order by cit.fechaHora desc;
end //
delimiter ;


-- SP 26: Insumos por debajo del stock mínimo con cantidad sugerida a comprar

create procedure sp_ReporteReabastecimiento()

    select

        nombreInsumo,
        stockActual,

-- SP 27: Certificados próximos a vencer (por defecto 15 días)
delimiter //
create procedure sp_CertificadosPorVencer(in p_dias int)
begin
    select
        cert.idCertificado,
        cli.nombreCliente,
        cli.celularCliente,
        cli.correoCliente,
        cert.fechaVencimiento,
        datediff(cert.fechaVencimiento, curdate()) as diasRestantes
    from CertificadoFumigacion cert
    inner join ServicioDirecto s on cert.idServicio = s.idServicio
    inner join CitaFumigacion cit on s.idCita = cit.idCita
    inner join Propiedad pr on cit.idPropiedad = pr.idPropiedad
    inner join Clientes cli on pr.idCliente = cli.idCliente
    where cert.estadoVigencia = 'Vigente'
      and cert.fechaVencimiento between curdate() and date_add(curdate(), interval coalesce(p_dias, 15) day)
    order by cert.fechaVencimiento;
end //
delimiter ;


-- SP 28: Ganancia neta en un rango de fechas
-- (servicios cobrados - órdenes de compra recibidas)
delimiter //
create procedure sp_GananciaNeta(
    in p_fechaInicio date,
    in p_fechaFin date
)
begin
    declare v_ingresos decimal(14,2);
    declare v_compras decimal(14,2);

    select coalesce(sum(costoTotal), 0) into v_ingresos
    from ServicioDirecto
    where estadoServicio = true
      and date(fechaTransaccion) between p_fechaInicio and p_fechaFin;

    select coalesce(sum(valorTotal), 0) into v_compras
    from OrdenCompra
    where estadoCompra = 'Recibida'
      and fechaCompra between p_fechaInicio and p_fechaFin;

    select p_fechaInicio as desde, p_fechaFin as hasta,
           v_ingresos as ingresosServicios,
           v_compras as costoCompras,
           v_ingresos - v_compras as gananciaNeta;
end //
delimiter ;


-- SP 29: Ranking Top 5 de plagas y de tipos de inmueble más atendidos en un mes
delimiter //
create procedure sp_RankingServicios(in p_anio int, in p_mes int)
begin
    select p.nombrePlaga, count(*) as serviciosRealizados, sum(s.costoTotal) as ingresos
    from ServicioDirecto s
    inner join Plaga p on s.idPlaga = p.idPlaga
    where year(s.fechaTransaccion) = p_anio and month(s.fechaTransaccion) = p_mes
    group by p.nombrePlaga
    order by serviciosRealizados desc, ingresos desc
    limit 5;

    select pr.tipoPropiedad, count(*) as serviciosRealizados, sum(s.costoTotal) as ingresos
    from ServicioDirecto s
    inner join CitaFumigacion cit on s.idCita = cit.idCita
    inner join Propiedad pr on cit.idPropiedad = pr.idPropiedad
    where year(s.fechaTransaccion) = p_anio and month(s.fechaTransaccion) = p_mes
    group by pr.tipoPropiedad
    order by serviciosRealizados desc, ingresos desc
    limit 5;
end //
delimiter ;

-- ---------- INVENTARIO ----------

-- TRIGGER 1: No permitir ajustes que dejen el stock en negativo
delimiter //
create trigger trg_ValidarStockAjuste
before insert on AjusteInventario
for each row
begin
    declare v_stock decimal(10,2);

    if new.tipoAjuste in ('Salida', 'Consumo Servicio', 'Merma') then
        select stockActual into v_stock
        from InsumoQuimico
        where idInsumoQuimico = new.idInsumoQuimico;

        if v_stock < new.cantidad then
            signal sqlstate '45000'
            set MESSAGE_TEXT = 'Error de Negocio: Stock insuficiente para realizar el ajuste.';
        end if;
    end if;
end //
delimiter ;


-- TRIGGER 2: Cada ajuste mueve el stock (suma o resta según el tipo)
delimiter //
create trigger trg_AplicarAjusteStock
after insert on AjusteInventario
for each row
begin
    if new.tipoAjuste in ('Entrada', 'Compra Recibida') then
        update InsumoQuimico
        set stockActual = stockActual + new.cantidad
        where idInsumoQuimico = new.idInsumoQuimico;
    else
        update InsumoQuimico
        set stockActual = stockActual - new.cantidad
        where idInsumoQuimico = new.idInsumoQuimico;
    end if;
end //
delimiter ;


-- TRIGGER 3: El estado del insumo se actualiza solo según su stock
delimiter //
create trigger trg_ActualizarEstadoInsumo
before update on InsumoQuimico
for each row
begin
    if new.estadoInsumo in ('Disponible', 'Stock Bajo', 'Agotado') then
        if new.stockActual = 0 then
            set new.estadoInsumo = 'Agotado';
        elseif new.stockActual <= new.stockMinimo then
            set new.estadoInsumo = 'Stock Bajo';
        else
            set new.estadoInsumo = 'Disponible';
        end if;
    end if;
end //
delimiter ;


-- TRIGGER 4: Auditoría de cambios de stock, precio y estado de insumos
delimiter //
create trigger trg_AuditoriaInsumoUpdate
after update on InsumoQuimico
for each row
begin
    if old.stockActual <> new.stockActual then
        insert into AuditoriaInventario (idRegistro, accion, valorAnterior, valorNuevo, idUsuario)
        values (new.idInsumoQuimico, 'CAMBIO_STOCK',
                concat('stock=', old.stockActual), concat('stock=', new.stockActual),
                coalesce(@idUsuarioActual, 1));
    end if;

    if old.precioListaActual <> new.precioListaActual then
        insert into AuditoriaInventario (idRegistro, accion, valorAnterior, valorNuevo, idUsuario)
        values (new.idInsumoQuimico, 'CAMBIO_PRECIO',
                concat('precio=', old.precioListaActual), concat('precio=', new.precioListaActual),
                coalesce(@idUsuarioActual, 1));
    end if;

    if old.estadoInsumo <> new.estadoInsumo then
        insert into AuditoriaInventario (idRegistro, accion, valorAnterior, valorNuevo, idUsuario)
        values (new.idInsumoQuimico, 'CAMBIO_ESTADO',
                concat('estado=', old.estadoInsumo), concat('estado=', new.estadoInsumo),
                coalesce(@idUsuarioActual, 1));
    end if;
end //
delimiter ;


-- ---------- SERVICIOS ----------

-- TRIGGER 5: Validar el servicio (cita válida, de tipo servicio directo y cliente activo)
delimiter //
create trigger trg_ValidarServicio
before insert on ServicioDirecto
for each row
begin
    declare v_estadoCita varchar(20);
    declare v_tipo varchar(50);
    declare v_estadoCliente varchar(20);

    select cit.estadoCita, cit.tipoSolicitud, cli.estadoCliente
    into v_estadoCita, v_tipo, v_estadoCliente
    from CitaFumigacion cit
    inner join Propiedad pr on cit.idPropiedad = pr.idPropiedad
    inner join Clientes cli on pr.idCliente = cli.idCliente
    where cit.idCita = new.idCita;

    if v_estadoCita is null then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: La cita indicada no existe.';
    end if;
    if v_tipo <> 'Servicio Directo' then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: La cita es de visita de cotización; agende una cita de servicio directo.';
    end if;
    if v_estadoCita = 'Cancelado' then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: No se puede registrar un servicio sobre una cita cancelada.';
    end if;
    if v_estadoCliente <> 'Activo' then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: El cliente está inactivo.';
    end if;
end //
delimiter ;


-- TRIGGER 6: Descuento automático de insumos al ejecutar un servicio

        insert into AjusteInventario (fechaAjuste, tipoAjuste, cantidad, justificacion, detalleAjuste, idInsumoQuimico)

            concat('Descuento automático por servicio #', new.idServicio),
            concat('Plaga tratada: ', p.nombrePlaga),
            p.idInsumoQuimico
        from Plaga p
        where p.idPlaga = new.idPlaga;
    end if;
end //
delimiter ;

-- ---------- CITAS ----------

-- TRIGGER 11: Validar cita nueva (cliente y propiedad activos, fecha futura,
-- técnico activo y sin cruce de horario con otra cita)
delimiter //
create trigger trg_ValidarCita
before insert on CitaFumigacion
for each row
begin
    declare v_estadoCliente varchar(20);
    declare v_estadoPropiedad varchar(20);
    declare v_estadoUsuario varchar(20);
    declare v_rolUsuario varchar(30);

    select cli.estadoCliente, p.estadoPropiedad into v_estadoCliente, v_estadoPropiedad
    from Propiedad p
    inner join Clientes cli on p.idCliente = cli.idCliente
    where p.idPropiedad = new.idPropiedad;

    if v_estadoCliente is null or v_estadoCliente <> 'Activo' or v_estadoPropiedad <> 'Activa' then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: La propiedad no existe o el cliente está inactivo.';
    end if;

    if new.fechaHora < now() then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: No se puede agendar una cita en el pasado.';
    end if;

    select estadoUsuario, rolUsuario into v_estadoUsuario, v_rolUsuario
    from Usuarios where idUsuario = new.idUsuario;
    if v_estadoUsuario is null or v_estadoUsuario <> 'Activo' or v_rolUsuario not in ('Técnico', 'Administrador') then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: El responsable de la cita debe ser un técnico activo.';
    end if;

    if not fn_TecnicoDisponible(new.idUsuario, new.fechaHora, new.duracionEstimadaMin, null) then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: El técnico ya tiene una cita en ese rango de horario.';
    end if;
end //
delimiter ;


-- TRIGGER 12: Al agendar una visita de cotización se crea su registro en $0
delimiter //
create trigger trg_CrearVisitaPendiente
after insert on CitaFumigacion
for each row

    if new.tipoSolicitud = 'Visita Cotización' then
        insert into VisitaCotizacion (valorServicio, areaMedidaM2, estadoCotizacion, idCita)
        values (0.00, null, 'Por inspeccionar', new.idCita);
    end if;
end //
delimiter ;


-- TRIGGER 13: Controlar cambios de la cita
--  - Una cita Realizada o Cancelada ya no cambia de estado.
--  - Solo pasa a 'Realizado' si tiene servicio registrado o visita cotizada.
--  - Reprogramar: solo citas pendientes/en camino, a futuro y sin cruces.
delimiter //
create trigger trg_ValidarCambioCita
before update on CitaFumigacion
for each row
begin
    if old.estadoCita in ('Realizado', 'Cancelado') and new.estadoCita <> old.estadoCita then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: Una cita realizada o cancelada no puede cambiar de estado.';
    end if;

    if new.estadoCita = 'Realizado' and old.estadoCita <> 'Realizado' then
        if new.tipoSolicitud = 'Servicio Directo'
           and not exists (select 1 from ServicioDirecto where idCita = new.idCita) then
            signal sqlstate '45000'
            set MESSAGE_TEXT = 'Error de Negocio: No se puede cerrar la cita sin registrar el servicio (plaga y m2).';
        end if;
        if new.tipoSolicitud = 'Visita Cotización'
           and not exists (select 1 from VisitaCotizacion where idCita = new.idCita and estadoCotizacion = 'Cotizada') then
            signal sqlstate '45000'
            set MESSAGE_TEXT = 'Error de Negocio: No se puede cerrar la visita sin registrar el área medida y el valor.';
        end if;
    end if;

    if new.fechaHora <> old.fechaHora or new.duracionEstimadaMin <> old.duracionEstimadaMin then
        if old.estadoCita not in ('Pendiente', 'En Camino') then
            signal sqlstate '45000'
            set MESSAGE_TEXT = 'Error de Negocio: Solo se pueden reprogramar citas pendientes o en camino.';
        end if;
        if new.fechaHora < now() then
            signal sqlstate '45000'
            set MESSAGE_TEXT = 'Error de Negocio: No se puede reprogramar a una fecha pasada.';
        end if;
        if not fn_TecnicoDisponible(new.idUsuario, new.fechaHora, new.duracionEstimadaMin, new.idCita) then
            signal sqlstate '45000'
            set MESSAGE_TEXT = 'Error de Negocio: El técnico no está disponible en el nuevo horario.';
        end if;
    end if;
end //
delimiter ;


-- TRIGGER 14: Solo las citas de cotización pueden tener visita

    end if;
end //

after update on Clientes

        and old.celularCliente <=> new.celularCliente
        and old.telefonoCliente <=> new.telefonoCliente

                       '; correo=', old.correoCliente, '; estado=', old.estadoCliente),
                concat('nombre=', new.nombreCliente, '; celular=', new.celularCliente, '; telefono=', coalesce(new.telefonoCliente, ''),

-- TRIGGER 24: Validar cambios de estado de la orden de compra
delimiter //
create trigger trg_ValidarEstadoOrden
before update on OrdenCompra
for each row
begin
    if old.estadoCompra in ('Recibida', 'Cancelada') and new.estadoCompra <> old.estadoCompra then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: Una orden recibida o cancelada no puede cambiar de estado.';

    if new.estadoCompra = 'Recibida' and old.estadoCompra <> 'Recibida'
       and not exists (select 1 from DetalleOrdenCompra where idCompra = new.idCompra) then
        signal sqlstate '45000'
        set MESSAGE_TEXT = 'Error de Negocio: No se puede recibir una orden sin insumos.';

create trigger trg_RecibirOrdenCompra
after update on OrdenCompra
for each row

        insert into AjusteInventario (fechaAjuste, tipoAjuste, cantidad, justificacion, detalleAjuste, idInsumoQuimico)

            'Compra Recibida',

            concat('Ingreso por orden de compra #', new.idCompra),
            concat('Precio unitario: ', d.precioUnitario),
            d.idInsumoQuimico
        from DetalleOrdenCompra d
        where d.idCompra = new.idCompra;
    end if;
end //
